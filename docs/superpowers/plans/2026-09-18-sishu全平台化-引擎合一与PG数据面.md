```python
# backend/app/services/sishu_data/binding.py（步骤3：最少实现）
from fastapi import HTTPException, Request
from ..core.auth import ENABLE_AUTH, get_current_user

def sishu_user_binding(u: str | None, request: Request) -> str:
    """?u= 会话绑定单点（协议 F②/R3-C2）。auth=1 非 admin：u 必须等于会话用户绑定 slug
    （认两形：username / h5_{username}）；auth=0 或 admin：桌面语义（缺省=admin）。
    用户属性名以 app/core/auth.py 用户模型为准（is_admin()/username——⑥-2a 批16 同款用法）。"""
    if not ENABLE_AUTH:
        return u or "admin"
    user = get_current_user(request)
    if user.is_admin():
        return u or "admin"
    name = (getattr(user, "username", "") or "").strip().lower()
    if not u or u not in (name, f"h5_{name}"):
        raise HTTPException(status_code=403, detail="未获该用户数据授权")
    return u
```

// frontend/src/pages/tutor/fetchBearerPatch.ts —— 传输层豁免类：契约/页面/选择器零改
import { getStoredToken, isAuthEnabled } from "../../auth/oidc";

let installed = false;
export function installFetchBearerPatch(): void {
  if (installed || typeof window === "undefined") return;
  installed = true;
  const _origFetch = window.fetch.bind(window);
  window.fetch = (input: RequestInfo | URL, init?: RequestInit) => {
    if (isAuthEnabled()) {                       // auth=0 惰性：零注入零影响
      const t = getStoredToken();
      if (t?.access_token) {
        const headers = new Headers(init?.headers);
        if (!headers.has("Authorization")) headers.set("Authorization", `Bearer ${t.access_token}`);
        init = { ...init, headers };
      }
    }
    return _origFetch(input, init);
  };
}
