```python
def test_probe_cached_within_ttl(monkeypatch):
    from app.services import engine_health as H
    calls = []
    monkeypatch.setattr(H, "_probe_doris", lambda: calls.append(1))
    H.invalidate("doris")
    H.probe("doris"); H.probe("doris")           # 第二次命中缓存
    assert len(calls) == 1
    H.probe("doris", force=True)                  # 穿透
    assert len(calls) == 2
```
