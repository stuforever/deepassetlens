## 一、纪律

**不动清单**：m15 既有 8 端点语义（GET""/GET{id}/POST""/DELETE{id}/POST upload/DELETE documents/POST vectorize/POST search——api/knowledge_base.py L108-357）；`_chunk_text`/`embed_texts`/TupuQdrantClient 基建；**工单域**（biz_work_order 两端点+其测试——零触碰回归底线）；向量管理页既有选择器；①②交付物。

**行为契约（spec §七）**：Qdrant 不可用→vectorize 文档行 error+search 可读错误（不静默空结果）；解析失败→行 failed 不阻同批；指针失联→报告 unreachable+不自动删注册；重嵌永远手动按钮。**指针二分（D6）**：connected 删除**只删注册行**；级联删（collection/磁盘目录/文档行）**仅 indexed 执行**——m15 现删库语义=indexed 专属。

**铁律**：建库表单白名单（type ∈ {indexed, connected}；rag_provider ∈ {qdrant, connected_es}；indexed 的 pointer_params 必空 422）；status 枚举扩展（stale/degraded）为应用层语义（列本就是 String，无 DDL）；解析器依赖先查（pdf/docx 没有的格式 422 明示）。

from typing import Any, Dict, List, Protocol, runtime_checkable


@runtime_checkable
class RAGPipeline(Protocol):
    """引擎族稳定接缝：initialize/add_documents/search/delete。"""

    def initialize(self, kb: Dict[str, Any]) -> None: ...

    def add_documents(self, kb: Dict[str, Any], docs: List[Dict[str, Any]]) -> Dict[str, int]:
        """docs=[{doc_id, filename, text}] → {doc_id: chunk_count}。"""

    def search(self, kb: Dict[str, Any], query: str, top_k: int = 6) -> List[Dict[str, Any]]:
        """→ [{text, score, payload}]。"""

    def delete(self, kb: Dict[str, Any], doc_ids: List[str] | None = None) -> None:
        """doc_ids=None 删整库；否则按文档删（Qdrant filter delete）。"""

def _ensure_kb_upgrade_columns(db) -> None:
    """④（spec §七 Runbook 步骤1）：两表加列，幂等；存量语义等值（现网全是索引型）。"""
    from sqlalchemy import text
    want = {"kg_knowledge_base": [
                ("type", "VARCHAR(20) NOT NULL DEFAULT 'indexed'"),
                ("rag_provider", "VARCHAR(40) NOT NULL DEFAULT 'qdrant'"),
                ("pointer_params", "JSON NULL"),
                ("embedding_signature", "JSON NULL"),
                ("enabled", "TINYINT(1) NOT NULL DEFAULT 1"),
                ("version", "INT NOT NULL DEFAULT 1")],
            "kg_knowledge_document": [
                ("checksum", "VARCHAR(64) NULL"),
                ("parse_cache_key", "VARCHAR(128) NULL")]}
    for table, cols in want.items():
        have = {r[0] for r in db.execute(text(
            "SELECT column_name FROM information_schema.columns "
            "WHERE table_schema=DATABASE() AND table_name=:t"), {"t": table})}
        for name, ddl in cols:
            if name not in have:
                db.execute(text(f"ALTER TABLE {table} ADD COLUMN {name} {ddl}"))
    db.commit()

"""④（spec D6）：connected 指针族——不复制、不重建、删除永不碰外部资源。
检索透传（ES _search→结果归一化为 [{text, score, payload}]）；add_documents=显式拒绝
（指针型无摄入语义——管理页上传入口对 connected 隐藏+后端 422 双保险）。"""
from typing import Any, Dict, List


class ConnectedESFamily:
    def __init__(self, params: Dict[str, Any]):
        self.url = params.get("url")            # 如 http://127.0.0.1:11200
        self.index = params.get("index", "")
        self._client = params.get("client")     # 可注入（测试桩）

    def initialize(self, kb: Dict[str, Any]) -> None:
        pass                                    # 指针不建索引——连通性由状态报告探测

    def add_documents(self, kb, docs) -> Dict[str, int]:
        raise ValueError("指针型 KB 不支持文档摄入（不复制不重建）")

    def search(self, kb, query: str, top_k: int = 6) -> List[Dict[str, Any]]:
        import requests
        body = {"query": {"match": {"_all_或指定字段": query}}, "size": top_k}
        resp = (self._client or requests).post(f"{self.url}/{self.index}/_search", json=body, timeout=10)
        hits = resp.json().get("hits", {}).get("hits", [])
        return [{"text": str(h.get("_source", {}))[:500], "score": h.get("_score", 0),
                 "payload": {"_id": h.get("_id"), "_index": h.get("_index")}} for h in hits]

    def delete(self, kb, doc_ids=None) -> None:
        return None                             # 纪律：删除零触碰外部（A3 断言钉死）

@router.post("/{kb_id}/reconcile")
def reconcile_knowledge_base(kb_id: str, db: Session = Depends(get_db)):
    """对账（spec D8）：文档行签名 vs KB 级签名（嵌入模型名/维度/分块参数三因子）→
    不一致标 stale；返回报告 {stale:[], failed:[], consistent:n, signature, checked_at}。
    永不自动重嵌（成本纪律）——重嵌走 reembed 按钮。幂等（重复对账结果稳定）。"""
    kb = _must_get_kb(kb_id, db)
    sig = kb.embedding_signature or _current_signature(db)      # 缺省=当前配置即时算
    stale = failed = consistent = 0
    for doc in kb.documents:
        if doc.status == "error":
            failed += 1
        elif (doc.status == "vectorized") and (doc.embedding_signature != sig):
            doc.status = "stale"; stale += 1
        else:
            consistent += 1
    kb.status = "degraded" if (stale or failed) else kb.status  # 状态报告纪律
    db.commit(); db.refresh(kb)
    return {"code": 200, "data": _kb_to_dict(kb, with_docs=True) | {"reconcile": {...}}}
